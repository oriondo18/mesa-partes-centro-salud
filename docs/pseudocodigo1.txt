INICIO registrar_paciente
    PEDIR apellido al usuario
    PEDIR dni al usuario

    SI validar_dni(dni) es verdadero ENTONCES
        historia = cantidad de elementos de la lista historias + 1
        AGREGAR historia a la lista historias
        AGREGAR apellido a la lista apellidos
        AGREGAR dni a la lista dnis
        LLAMAR ordenar_pacientes()
        MOSTRAR "Paciente registrado con historia", historia
    SINO
        MOSTRAR "DNI invalido, no se registro"
    FIN SI
FIN


INICIO buscar_paciente
    PEDIR numero de historia al usuario 
    encontrado = falso

    SI el texto contiene solo dígitos ENTONCES
        historia = convertir el texto a número entero

        PARA cada posición i de la lista historias HACER
            SI historias[i] es igual a historia ENTONCES
                MOSTRAR "Paciente encontrado"
                MOSTRAR N° HC: historias[i]
                MOSTRAR apellidos[i]
                MOSTRAR dnis[i]
                encontrado = verdadero
            FIN SI
        FIN PARA
    FIN SI

    SI encontrado es falso ENTONCES
        MOSTRAR "Paciente no encontrado"
    FIN SI
FIN
